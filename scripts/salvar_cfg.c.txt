/* ----- salvar uma copia do .cfg com as posicoes/tamanhos ajustados ----- */
#include <streams/file_stream.h>
#define MOD_CFG_MARCA "# criado pelo RetroArch Mod (editor de overlay)"
static float      *mod_snap    = NULL;   /* 4 floats por botao (x, y, range_x, range_y) do momento da carga */
static size_t      mod_snap_n  = 0;
static const void *mod_snap_id = NULL;
static char        mod_snap_path[PATH_MAX_LENGTH];
static char        mod_last_saved[PATH_MAX_LENGTH];
static char        mod_main_name[64];             /* pagina principal (nome); vai no .cfg como mod_main_page */
static char        mod_origin[PATH_MAX_LENGTH];   /* .cfg de onde veio o arquivo temporario */

static void input_overlay_free(input_overlay_t *ol);

/* descarta o overlay guardado em cache se for o arquivo que acabamos de gravar (senao ele voltaria velho) */
static void mod_drop_cache(const char *path)
{
   input_overlay_t *c = input_driver_st.overlay_cache_ptr;
   if (c && c->path && !strcmp(c->path, path))
   {
      input_overlay_free(c);
      input_driver_st.overlay_cache_ptr = NULL;
   }
}

static bool mod_is_temp(const char *path)
{
   const char *b = path_basename(path);
   return !strcmp(b, "_mod_temp_.cfg");
}

/* o .cfg "de verdade" que o usuario esta editando (o temporario aponta para a origem) */
static const char *mod_eff_path(const char *loaded)
{
   return (mod_is_temp(loaded) && mod_origin[0]) ? mod_origin : loaded;
}

static void mod_toast(const char *m)
{
   runloop_msg_queue_push(m, strlen(m), 1, 480, true, NULL,
         MESSAGE_QUEUE_ICON_DEFAULT, MESSAGE_QUEUE_CATEGORY_INFO);
}

/* procura a linha: mod_main_page = "nome" */
static bool mod_find_main(const char *buf, size_t len, char *out, size_t cap)
{
   const char *p = buf, *end = buf + len;
   while (p < end)
   {
      const char *eol = (const char*)memchr(p, '\n', (size_t)(end - p));
      size_t      n   = eol ? (size_t)(eol - p) : (size_t)(end - p);
      if (n > 13 && !strncmp(p, "mod_main_page", 13))
      {
         char t[256];
         const char *q;
         size_t m = n < sizeof(t) - 1 ? n : sizeof(t) - 1;
         memcpy(t, p, m);
         t[m] = '\0';
         q = strchr(t, '=');
         if (q)
         {
            size_t l;
            q++;
            while (*q == ' ' || *q == '\t' || *q == '"')
               q++;
            strlcpy(out, q, cap);
            l = strlen(out);
            while (l > 0 && (out[l - 1] == ' ' || out[l - 1] == '\r' || out[l - 1] == '"'))
               out[--l] = '\0';
            return out[0] != '\0';
         }
      }
      p += eol ? n + 1 : n;
   }
   return false;
}

/* Guarda os valores originais de todas as paginas (uma vez por overlay carregado). */
static void mod_snap_take(input_overlay_t *ol)
{
   size_t i, j, k = 0, n = 0;
   if (!ol || !ol->overlays)
      return;
   for (i = 0; i < ol->size; i++)
      n += ol->overlays[i].size;
   if (   mod_snap && mod_snap_id == (const void*)ol->overlays
       && mod_snap_n == n && !strcmp(mod_snap_path, ol->path))
      return;
   free(mod_snap);
   mod_snap   = (float*)malloc((n ? n : 1) * 4 * sizeof(float));
   mod_snap_n = 0;
   if (!mod_snap)
      return;
   for (i = 0; i < ol->size; i++)
      for (j = 0; j < ol->overlays[i].size; j++, k++)
      {
         const struct overlay_desc *d = &ol->overlays[i].descs[j];
         mod_snap[k * 4 + 0] = d->x;
         mod_snap[k * 4 + 1] = d->y;
         mod_snap[k * 4 + 2] = d->range_x;
         mod_snap[k * 4 + 3] = d->range_y;
      }
   mod_snap_n  = n;
   mod_snap_id = ol->overlays;
   strlcpy(mod_snap_path, ol->path, sizeof(mod_snap_path));
   /* pagina principal gravada no .cfg */
   mod_main_name[0] = '\0';
   {
      void   *mb = NULL;
      int64_t ml = 0;
      if (ol->path[0] && filestream_read_file(ol->path, &mb, &ml) && mb && ml > 0)
         mod_find_main((const char*)mb, (size_t)ml, mod_main_name, sizeof(mod_main_name));
      free(mb);
   }
}

/* Ao carregar qualquer overlay (jogo, troca de .cfg): vai para a pagina marcada como principal */
static void mod_apply_main_page(input_overlay_t *ol)
{
   void   *mb = NULL;
   int64_t ml = 0;
   char    nm[64];
   size_t  i;
   bool    tmp = false;
   if (!ol || !ol->overlays || !ol->path[0])
      return;
   if (!filestream_read_file(ol->path, &mb, &ml) || !mb || ml <= 0)
   {
      free(mb);
      return;
   }
   nm[0] = '\0';
   if (mod_find_main((const char*)mb, (size_t)ml, nm, sizeof(nm)))
      for (i = 0; i < ol->size; i++)
         if (!strcmp(ol->overlays[i].name, nm))
         {
            if (i != (size_t)ol->index)
            {
               ol->next_index = (unsigned)i;
               command_event(CMD_EVENT_OVERLAY_NEXT, &tmp);
            }
            break;
         }
   free(mb);
}

static bool mod_cfg_is_mod(const char *path)
{
   void   *b   = NULL;
   int64_t len = 0;
   bool    r;
   if (!path || !path[0] || !filestream_read_file(path, &b, &len) || !b)
      return false;
   r = (len >= (int64_t)strlen(MOD_CFG_MARCA)
         && !strncmp((const char*)b, MOD_CFG_MARCA, strlen(MOD_CFG_MARCA)));
   free(b);
   return r;
}

/* nome novo para uma copia: <base>-mod.cfg, -mod2.cfg, ... (primeiro que nao existe) */
static void mod_new_dst(const char *src, char *dst, size_t cap)
{
   char     base[PATH_MAX_LENGTH];
   size_t   bl;
   unsigned n;
   strlcpy(base, src, sizeof(base));
   bl = strlen(base);
   if (bl > 4 && !strcasecmp(base + bl - 4, ".cfg"))
      base[bl - 4] = '\0';
   bl = strlen(base);
   {
      /* tira um "-mod" ou "-modN" do final, para nao acumular */
      size_t q = bl;
      while (q > 0 && base[q - 1] >= '0' && base[q - 1] <= '9')
         q--;
      if (q >= 4 && !strncmp(base + q - 4, "-mod", 4))
         base[q - 4] = '\0';
   }
   for (n = 1; n < 1000; n++)
   {
      if (n == 1)
         snprintf(dst, cap, "%s-mod.cfg", base);
      else
         snprintf(dst, cap, "%s-mod%u.cfg", base, n);
      if (!path_is_valid(dst))
         break;
   }
}

/* Texto do .cfg com as linhas dos botoes movidos/redimensionados trocadas
 * (comparando com os valores do momento da carga) e a marca do app no inicio.
 * Devolve memoria alocada (free) ou NULL. */
static char *mod_apply_moves(input_overlay_t *ol, const char *buf, size_t len,
      size_t *olen_out, size_t *changed_out, char *ex, size_t excap)
{
   const char *p = buf, *end = buf + len;
   char   *out;
   size_t  cap, olen = 0, i, nl = 1, changed = 0;
   bool    has_mark = (len >= strlen(MOD_CFG_MARCA)
         && !strncmp(p, MOD_CFG_MARCA, strlen(MOD_CFG_MARCA)));
   for (i = 0; i < len; i++)
      if (p[i] == '\n')
         nl++;
   cap = len + 80 * nl + 256;
   out = (char*)malloc(cap);
   if (!out)
      return NULL;
   if (ex)
      ex[0] = '\0';
   if (!has_mark)
      olen += (size_t)snprintf(out + olen, cap - olen, "%s\n", MOD_CFG_MARCA);
   while (p < end)
   {
      const char *eol = (const char*)memchr(p, '\n', (size_t)(end - p));
      size_t      n   = eol ? (size_t)(eol - p) : (size_t)(end - p);
      size_t      adv = eol ? n + 1 : n;
      bool        done = false;
      char        t[1024];
      if (mod_main_name[0] && n > 13 && !strncmp(p, "mod_main_page", 13))
      {
         p += adv;
         continue;
      }
      unsigned    o, d;
      int         nn = 0;
      if (n > 0 && n < sizeof(t))
      {
         memcpy(t, p, n);
         t[n] = '\0';
         if (sscanf(t, "overlay%u_desc%u%n", &o, &d, &nn) == 2 && nn > 0)
         {
            char *q1, *q2;
            size_t k = (size_t)nn;
            while (t[k] == ' ' || t[k] == '\t')
               k++;
            if (   t[k] == '=' && (q1 = strchr(t + k, '"'))
                && (q2 = strchr(q1 + 1, '"'))
                && o < ol->size && d < ol->overlays[o].size)
            {
               char   v[512];
               char  *tok[8];
               int    nt = 0;
               char  *c;
               size_t vl = (size_t)(q2 - (q1 + 1));
               if (vl < sizeof(v))
               {
                  memcpy(v, q1 + 1, vl);
                  v[vl] = '\0';
                  c = v;
                  tok[nt++] = c;
                  while (nt < 8 && (c = strchr(c, ',')))
                  {
                     *c++ = '\0';
                     tok[nt++] = c;
                  }
                  if (nt == 6)
                  {
                     size_t off = 0, pg;
                     const struct overlay_desc *dd = &ol->overlays[o].descs[d];
                     const float *sv;
                     for (pg = 0; pg < o; pg++)
                        off += ol->overlays[pg].size;
                     sv = &mod_snap[(off + d) * 4];
                     if (   fabsf(dd->x - sv[0]) > 1e-6f
                         || fabsf(dd->y - sv[1]) > 1e-6f
                         || fabsf(dd->range_x - sv[2]) > 1e-6f
                         || fabsf(dd->range_y - sv[3]) > 1e-6f)
                     {
                        size_t pre = (size_t)(q1 + 1 - t);
                        memcpy(out + olen, t, pre);
                        olen += pre;
                        olen += (size_t)snprintf(out + olen, cap - olen,
                              "%s,%.5f,%.5f,%s,%.6f,%.6f",
                              tok[0], dd->x, dd->y, tok[3],
                              dd->range_x, dd->range_y);
                        memcpy(out + olen, q2, n - (size_t)(q2 - t));
                        olen += n - (size_t)(q2 - t);
                        if (eol)
                           out[olen++] = '\n';
                        if (ex && !ex[0])
                           snprintf(ex, excap, "; ex.: pag %u botao %u x %.3f>%.3f y %.3f>%.3f",
                                 o, d, sv[0], dd->x, sv[1], dd->y);
                        changed++;
                        done = true;
                     }
                  }
               }
            }
         }
      }
      if (!done)
      {
         memcpy(out + olen, p, adv);
         olen += adv;
      }
      p += adv;
   }
   if (mod_main_name[0])
   {
      if (olen > 0 && out[olen - 1] != '\n')
         out[olen++] = '\n';
      olen += (size_t)snprintf(out + olen, cap - olen, "mod_main_page = \"%s\"\n", mod_main_name);
   }
   *olen_out    = olen;
   *changed_out = changed;
   return out;
}

static void mod_save_cfg(bool overwrite)
{
   input_overlay_t *ol = input_driver_st.overlay_ptr;
   char     dst[PATH_MAX_LENGTH];
   char     msg[PATH_MAX_LENGTH + 96];
   void    *buf  = NULL;
   int64_t  len  = 0;
   char    *out;
   size_t   olen = 0, changed = 0;
   char     ex[128];
   if (!ol || !ol->overlays || !mod_snap || !ol->path[0])
   {
      mod_toast("Salvar: overlay nao carregado.");
      return;
   }
   if (!filestream_read_file(ol->path, &buf, &len) || !buf || len <= 0)
   {
      mod_toast("Salvar: nao consegui ler o .cfg do overlay.");
      free(buf);
      return;
   }
   dst[0] = '\0';
   if (overwrite)
   {
      if (mod_cfg_is_mod(mod_eff_path(ol->path)))
         strlcpy(dst, mod_eff_path(ol->path), sizeof(dst));
      else if (mod_last_saved[0] && mod_cfg_is_mod(mod_last_saved))
         strlcpy(dst, mod_last_saved, sizeof(dst));
      else
      {
         mod_toast("Sobrescrever so vale para copia criada pelo app. Use Salvar novo.");
         free(buf);
         return;
      }
   }
   else
      mod_new_dst(mod_eff_path(ol->path), dst, sizeof(dst));
   out = mod_apply_moves(ol, (const char*)buf, (size_t)len, &olen, &changed, ex, sizeof(ex));
   if (!out)
   {
      free(buf);
      return;
   }
   if (filestream_write_file(dst, out, (int64_t)olen))
   {
      void   *chk  = NULL;
      int64_t clen = 0;
      bool    ok   = filestream_read_file(dst, &chk, &clen) && chk
            && (size_t)clen == olen && !memcmp(chk, out, olen);
      free(chk);
      strlcpy(mod_last_saved, dst, sizeof(mod_last_saved));
      mod_drop_cache(dst);
      snprintf(msg, sizeof(msg), "Salvo: %s a partir de %s (%u botoes alterados%s)%s%s",
            path_basename(dst), path_basename(mod_eff_path(ol->path)), (unsigned)changed, ex,
            ok ? "" : " [conferencia do arquivo FALHOU]",
            mod_main_name[0] ? " Pagina principal gravada." : "");
   }
   else
      snprintf(msg, sizeof(msg), "Salvar: falhou ao gravar %s", path_basename(dst));
   mod_toast(msg);
   free(out);
   free(buf);
}

/* ----- gerenciar copias criadas pelo app: renomear e apagar (chamado pelo menu) ----- */
void input_overlay_mod_toast(const char *m)
{
   mod_toast(m);
}

bool input_overlay_mod_is_copy(const char *path)
{
   return mod_cfg_is_mod(path);
}

/* true se o arquivo e o overlay carregado agora ou o escolhido nas configuracoes */
bool input_overlay_mod_in_use(const char *path)
{
   input_overlay_t *ol = input_driver_st.overlay_ptr;
   settings_t *st      = config_get_ptr();
   return (ol && ol->path[0] && !strcmp(ol->path, path))
       || (st && st->paths.path_overlay[0] && !strcmp(st->paths.path_overlay, path));
}

bool input_overlay_mod_delete(const char *path)
{
   char msg[PATH_MAX_LENGTH + 64];
   if (!mod_cfg_is_mod(path))
   {
      mod_toast("So da para apagar copias criadas pelo app.");
      return false;
   }
   if (input_overlay_mod_in_use(path))
   {
      mod_toast("Esta copia esta em uso. Escolha outro overlay antes de apagar.");
      return false;
   }
   if (filestream_delete(path) != 0)
   {
      mod_toast("Nao consegui apagar o arquivo.");
      return false;
   }
   if (!strcmp(mod_last_saved, path))
      mod_last_saved[0] = '\0';
   snprintf(msg, sizeof(msg), "Apagado: %s", path_basename(path));
   mod_toast(msg);
   return true;
}

bool input_overlay_mod_rename(const char *path, const char *novo)
{
   char   nome[64];
   char   dst[PATH_MAX_LENGTH];
   char   msg[PATH_MAX_LENGTH + 64];
   size_t n = 0;
   const char *p;
   const char *barra;
   if (!novo)
      return false;
   if (!mod_cfg_is_mod(path))
   {
      mod_toast("So da para renomear copias criadas pelo app.");
      return false;
   }
   if (input_overlay_mod_in_use(path))
   {
      mod_toast("Esta copia esta em uso. Escolha outro overlay antes de renomear.");
      return false;
   }
   for (p = novo; *p == ' '; p++) { }
   for (; *p && n < sizeof(nome) - 1; p++)
   {
      char c = *p;
      if (c == ' ')
         c = '-';
      if (   (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
          || (c >= '0' && c <= '9') || c == '-' || c == '_')
         nome[n++] = c;
      else if (c == '.' && !strcasecmp(p, ".cfg"))
         break;
      else
      {
         mod_toast("Nome invalido: use so letras, numeros, - e _");
         return false;
      }
   }
   while (n > 0 && nome[n - 1] == '-')
      n--;
   nome[n] = '\0';
   if (n == 0)
   {
      mod_toast("Nome vazio.");
      return false;
   }
   barra = strrchr(path, '/');
   if (!barra)
      return false;
   snprintf(dst, sizeof(dst), "%.*s/%s.cfg", (int)(barra - path), path, nome);
   if (!strcmp(dst, path))
      return false;
   if (path_is_valid(dst))
   {
      mod_toast("Ja existe um arquivo com esse nome.");
      return false;
   }
   if (filestream_rename(path, dst) != 0)
   {
      mod_toast("Nao consegui renomear o arquivo.");
      return false;
   }
   if (!strcmp(mod_last_saved, path))
      strlcpy(mod_last_saved, dst, sizeof(mod_last_saved));
   snprintf(msg, sizeof(msg), "Renomeado para: %s", path_basename(dst));
   mod_toast(msg);
   return true;
}
